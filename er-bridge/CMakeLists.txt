cmake_minimum_required(VERSION 3.25)
project(EldenCraftNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_subdirectory(third_party/minhook)
add_library(dinput8 SHARED src/proxy.cpp src/log.cpp src/shm.cpp src/paths.cpp src/dinput8.def)
add_library(erbridge_core SHARED src/core.cpp src/log.cpp src/shm.cpp src/paths.cpp
  src/memutil.cpp src/debugcmd.cpp src/frame.cpp src/game.cpp src/fps.cpp src/compositor_windows.cpp)
foreach(t dinput8 erbridge_core)
  target_include_directories(${t} PRIVATE include third_party/minhook/include)
  target_link_libraries(${t} PRIVATE user32 kernel32)
  target_compile_options(${t} PRIVATE /W3 /utf-8)
endforeach()
target_link_libraries(erbridge_core PRIVATE minhook)
add_executable(bridge_smoke tests/bridge_smoke.cpp src/shm.cpp src/log.cpp src/paths.cpp)
target_include_directories(bridge_smoke PRIVATE include src)
target_link_libraries(bridge_smoke PRIVATE user32 kernel32)
enable_testing()
add_executable(client_liveness_test tests/client_liveness_test.cpp)
target_include_directories(client_liveness_test PRIVATE src)
add_test(NAME client_liveness COMMAND client_liveness_test)
add_executable(memory_validation_test tests/memory_validation_test.cpp src/memutil.cpp src/log.cpp src/paths.cpp)
target_include_directories(memory_validation_test PRIVATE include src)
target_link_libraries(memory_validation_test PRIVATE user32 kernel32)
add_test(NAME memory_validation COMMAND memory_validation_test)
